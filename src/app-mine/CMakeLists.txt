add_library(mine_application STATIC
    core/mine.c kvm/app/app.c kvm/app/session.c
    kvm/text/scene.c kvm/graphics/scene.c kvm/io/number.c)
target_link_libraries(mine_application PUBLIC mygames_kvm)
target_include_directories(mine_application PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
set_target_properties(mine_application PROPERTIES C_STANDARD 90 C_STANDARD_REQUIRED YES C_EXTENSIONS NO)
option(MINE_COVERAGE "Instrument Mine's portable owners for local test review" OFF)
if(MINE_COVERAGE AND CMAKE_C_COMPILER_ID STREQUAL "GNU")
    target_compile_options(mine_application PRIVATE -O0 --coverage)
    target_link_options(mine_application INTERFACE --coverage)
endif()
if(NOT MSVC)
    target_compile_options(mine_application PRIVATE -Wall -Wextra -Werror -pedantic)
endif()
foreach(owner core application presentation)
    if(owner STREQUAL "core")
        set(owner_path core)
    else()
        set(owner_path kvm)
    endif()
    add_executable(mine_${owner}_test "${CMAKE_SOURCE_DIR}/test/app-mine/${owner_path}/${owner}_test.c")
    target_link_libraries(mine_${owner}_test PRIVATE mine_application)
    add_test(NAME mine.${owner} COMMAND mine_${owner}_test)
endforeach()
if(WIN32)
    add_executable(mine kvm/platform/win32/main.c)
    target_link_libraries(mine PRIVATE mine_application mygames_win32)
    if(CMAKE_SIZEOF_VOID_P EQUAL 8)
        set_target_properties(mine PROPERTIES OUTPUT_NAME mine64)
    else()
        set_target_properties(mine PROPERTIES OUTPUT_NAME mine32)
    endif()
    if(MINGW)
        target_link_options(mine PRIVATE -static)
    endif()
    if(NOT MSVC)
        target_compile_options(mine PRIVATE -Wall -Wextra -Werror -pedantic)
    endif()
    add_custom_target(mine_win32_host_integration
        COMMAND ${Python3_EXECUTABLE} "${CMAKE_SOURCE_DIR}/test/app-mine/kvm/platform/win32/integration.py"
            --product "$<TARGET_FILE:mine>" --output "${CMAKE_BINARY_DIR}/mine-host-integration"
        DEPENDS mine USES_TERMINAL VERBATIM)
endif()
